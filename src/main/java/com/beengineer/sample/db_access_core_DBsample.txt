■ データベース作成
--------------------------------------------------------------
CREATE DATABASE hrm
  WITH ENCODING 'UTF8'
       LC_COLLATE='C'
       LC_CTYPE='C'
       TEMPLATE template0;

■ 接続ユーザー作成
--------------------------------------------------------------
CREATE USER <<ユーザー名（任意）>>WITH PASSWORD 'xxxxx';
GRANT ALL PRIVILEGES ON DATABASE hrm TO <<ユーザー名（任意）>>;

■ 最小接続設定（/db-access-core/src/main/webapp/WEB-INF/system.xml）
--------------------------------------------------------------
<system>
  <database>
    <username>postgres</username>
    <password>パスワード（任意）</password>
    <!-- 接続URLは問題の無い形式で記述 -->
	<url>jdbc:postgresql://<<データベースサーバアドレス>>:5432/hrm</url>
    <connect>20</connect>
    <driver>org.postgresql.Driver</driver>
    <object_id>CTID</object_id>
  </database>
（省略）
</system>

■ サンプルテーブル（任意）
--------------------------------------------------------------
CREATE TABLE app_mst
(
  app_id char(3) NOT NULL,
  app_name text NOT NULL,
  app_kana text,
  create_dt timestamp NOT NULL DEFAULT now(),
  create_user char(8) NOT NULL,
  update_dt timestamp NOT NULL DEFAULT now(),
  update_user char(8) NOT NULL,
  del_flg char(1) NOT NULL DEFAULT '0'::bpchar,
  remark text,
  CONSTRAINT app_mst_pkey PRIMARY KEY (app_id)
);

ALTER TABLE app_mst OWNER TO postgres;

COMMENT ON TABLE app_mst IS 'アプリケーションマスタ';
